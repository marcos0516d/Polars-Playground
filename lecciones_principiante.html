📖 Guía de Funciones de Polars

import polars as pl

Importa la librería Polars y le asigna el alias pl. Es la convención estándar.

import polars as pl

print(pl.__version__)

pl.DataFrame()

Crea un DataFrame, la estructura principal de Polars. Se crea desde un diccionario de listas.

# DataFrame desde diccionario
df = pl.DataFrame({
    "producto": ["Laptop", "Mouse", "Teclado"],
    "precio": [1200, 25, 80],
    "cantidad": [5, 20, 15]
})

# Información del DataFrame
print(df.shape)      # (3, 3)
print(df.columns)    # Nombres de columnas
print(df.dtypes)     # Tipos de datos

pl.read_csv()

Lee un archivo CSV y lo convierte en un DataFrame de forma eager (inmediata).

# Leer un archivo CSV
df = pl.read_csv('ventas.csv')

# Con opciones
df = pl.read_csv(
    'datos.csv',
    separator=';',        # Delimitador
    has_header=True,      # Tiene encabezado
    truncate_ragged_lines=True
)

df.head()

Muestra las primeras filas de un DataFrame.

# Primeras 5 filas (por defecto)
print(df.head())

# Primeras N filas
print(df.head(10))

df.select()

Selecciona una o más columnas del DataFrame. Es la forma principal de seleccionar columnas en Polars.

# Seleccionar una columna
precios = df.select("precio")

# Seleccionar múltiples columnas
subset = df.select(["producto", "precio"])

# Seleccionar con expresiones
resumen = df.select(
    pl.col("precio").mean().alias("precio_promedio"),
    pl.col("cantidad").sum().alias("total_vendido")
)

df.filter()

Filtra filas que cumplan una condición usando expresiones pl.col().

# Filtrar por condición
caros = df.filter(pl.col("precio") > 100)

# Múltiples condiciones
resultado = df.filter(
    (pl.col("precio") > 50) & (pl.col("cantidad") > 10)
)

# Filtrar con is_in
seleccion = df.filter(pl.col("producto").is_in(["Laptop", "Teclado"]))

Expresiones con pl.col()

Las expresiones son la base de Polars. Permiten operar sobre columnas de forma eficiente.

# Operaciones matemáticas
df = df.with_columns(
    (pl.col("precio") * pl.col("cantidad")).alias("total")
)

# Alias para nombrar resultado
promedio = df.select(pl.col("precio").mean().alias("promedio"))

Operaciones estadísticas

Polars tiene métodos integrados para cálculos estadísticos.

# Promedio
promedio = df.select(pl.col("precio").mean())

# Suma total
total = df.select(pl.col("cantidad").sum())

# Máximo y mínimo
maximo = df.select(pl.col("precio").max())
minimo = df.select(pl.col("precio").min())

# Resumen completo
print(df.describe())

df.sort()

Ordena el DataFrame por una o más columnas.

# Ordenar de menor a mayor
ordenado = df.sort("precio")

# Ordenar de mayor a menor
top = df.sort("precio", descending=True)

# Ordenar por múltiples columnas
orden = df.sort(["cantidad", "precio"])

df.with_columns()

Agrega nuevas columnas al DataFrame sin modificar el original.

# Nueva columna con operación
df = df.with_columns(
    (pl.col("precio") * 1.19).alias("precio_con_iva")
)

# Múltiples columnas nuevas
df = df.with_columns([
    pl.col("precio").mean().alias("precio_promedio"),
    pl.col("cantidad").sum().alias("stock_total")
])

df.group_by().agg()

Agrupa filas por una columna y aplica agregaciones (sum, mean, count, etc.).

# Agrupar y sumar
totales = df.group_by("categoria").agg(
    pl.col("ventas").sum()
)

# Múltiples agregaciones
resumen = df.group_by("vendedor").agg([
    pl.col("monto").sum().alias("total"),
    pl.col("monto").mean().alias("promedio"),
    pl.col("monto").count().alias("n_ventas")
])

Valores faltantes

Polars maneja valores nulos de forma integrada.

# Eliminar filas con nulos
limpio = df.drop_nulls()

# Rellenar nulos con un valor
completo = df.fill_null(0)

# Rellenar nulos con la media
completo = df.fill_null(df.select(pl.col("cantidad").mean()))

df.write_csv()

Exporta el DataFrame a un archivo CSV.

# Guardar en CSV
df.write_csv("resultado.csv")

# Guardar sin índice
df.write_csv("resultado.csv")
