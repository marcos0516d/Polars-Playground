Lazy API: df.lazy() y collect()

Convierte un DataFrame en un LazyFrame. Las operaciones se acumulan como un plan de ejecución que se optimiza antes de ejecutarse con collect().

# Crear LazyFrame y ejecutar
resultado = (
    df.lazy()
    .filter(pl.col("ventas") > 100)
    .collect()
)

# Ver plan de ejecución
plan = df.lazy().filter(pl.col("ventas") > 100).explain()
print(plan)

Filtros con expresiones complejas

Combina múltiples condiciones con operadores & (AND) y | (OR).

# Filtro con AND
resultado = df.filter(
    (pl.col("cantidad") > 5) & (pl.col("precio") < 100)
)

# Filtro con OR
resultado = df.filter(
    (pl.col("categoria") == "A") | (pl.col("ventas") > 1000)
)

# Filtro con is_in
seleccion = df.filter(pl.col("producto").is_in(["A", "B", "C"]))

df.head() y df.slice()

Controla la cantidad de filas que retornas.

# Primeras N filas
primeras_5 = df.head(5)

# Slice: desde posición offset, tomar length filas
subset = df.slice(offset=2, length=3)  # Posiciones 2, 3, 4

Window functions con .over()

Calcula agregaciones por grupos sin colapsar filas. Se usa con .over().

# Promedio por grupo
df = df.with_columns(
    pl.col("ventas").mean().over("vendedor").alias("promedio_vendedor")
)

# Suma por categoría
df = df.with_columns(
    pl.col("monto").sum().over("categoria").alias("total_categoria")
)

# Múltiples agregaciones por grupo
df = df.with_columns([
    pl.col("ventas").mean().over("region").alias("promedio_region"),
    pl.col("ventas").max().over("region").alias("max_region"),
])

Rank con window

Asigna rankings dentro de grupos.

# Ranking dentro de cada vendedor
df = df.with_columns(
    pl.col("ventas").rank(descending=True).over("vendedor").alias("rank")
)

# Dense rank (sin saltos)
df = df.with_columns(
    pl.col("ventas").rank(method="dense", descending=True).over("vendedor").alias("dense_rank")
)

shift() y diff()

Accede a valores anteriores/siguientes o calcula diferencias.

# Valor anterior (shift por defecto es 1)
df = df.with_columns(
    pl.col("ventas").shift(1).alias("ventas_anterior")
)

# Diferencia con el valor anterior
df = df.with_columns(
    pl.col("ventas").diff().alias("diferencia")
)

# Shift negativo (valor siguiente)
df = df.with_columns(
    pl.col("ventas").shift(-1).alias("ventas_siguiente")
)

Rolling windows

Calcula medias móviles y otras agregaciones en ventanas deslizantes.

# Media móvil de 3 períodos
df = df.with_columns(
    pl.col("ventas").rolling_mean(window_size=3).alias("media_movil_3")
)

# Suma acumulada
df = df.with_columns(
    pl.col("ventas").cum_sum().alias("acumulado")
)

# Mínimo y máximo en ventana
df = df.with_columns([
    pl.col("ventas").rolling_min(window_size=3).alias("min_movil"),
    pl.col("ventas").rolling_max(window_size=3).alias("max_movil"),
])

SQL con Polars

Ejecuta consultas SQL sobre DataFrames usando el método sql().

# Registrar DataFrame y consultar con SQL
resultado = df.sql("SELECT * FROM self WHERE ventas > 100 ORDER BY ventas DESC")

# Consulta con GROUP BY
resumen = df.sql("""
    SELECT vendedor, SUM(ventas) as total
    FROM self
    GROUP BY vendedor
    ORDER BY total DESC
""")

CTEs en SQL

Common Table Expressions para consultas complejas.

# Usar CTEs con sql()
resultado = df.sql("""
    WITH top_vendedores AS (
        SELECT vendedor, SUM(ventas) as total
        FROM self
        GROUP BY vendedor
        HAVING total > 500
    )
    SELECT * FROM top_vendedores
    ORDER BY total DESC
""")

Expresiones complejas con alias, cast, when/then

Combina múltiples expresiones en una consulta.

# when/then (equivalente a CASE WHEN)
df = df.with_columns(
    pl.when(pl.col("ventas") > 1000)
    .then(pl.lit("Alto"))
    .when(pl.col("ventas") > 500)
    .then(pl.lit("Medio"))
    .otherwise(pl.lit("Bajo"))
    .alias("nivel")
)

# Cast de tipos
df = df.with_columns(
    pl.col("precio").cast(pl.Utf8).alias("precio_str")
)

Operaciones de texto con .str

Manipula strings con el accesor .str.

# Mayúsculas y minúsculas
df = df.with_columns(pl.col("producto").str.to_uppercase())

# Reemplazar texto
df = df.with_columns(pl.col("nombre").str.replace_all("old", "new"))

# Longitud
df = df.with_columns(pl.col("producto").str.len_bytes().alias("longitud"))

# Contiene
df = df.filter(pl.col("producto").str.contains("laptop"))

Categorical

Optimiza columnas con pocos valores únicos usando tipo categórico.

# Convertir a categorical
df = df.with_columns(pl.col("categoria").cast(pl.Categorical))

# Ordinal (con orden definido)
from polars import Enum
df = df.with_columns(
    pl.col("nivel").cast(Enum(["bajo", "medio", "alto"]))
)

explode() — Explorar listas

Expande una columna de listas en múltiples filas.

# Explode columnas de listas
df = pl.DataFrame({
    "producto": ["A", "B", "C"],
    "etiquetas": [["importante", "urgente"], ["normal"], ["urgente", "revision"]]
})

resultado = df.explode("etiquetas")
# A: importante, A: urgente, B: normal, C: urgente, C: revision
