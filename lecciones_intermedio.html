df.join() — Join por columna

Combina dos DataFrames basándose en una o más columnas comunes, similar a los JOINs en SQL.

# Join por una columna
empleados = pl.DataFrame({
    "nombre": ["Ana", "Bob", "Carlos"],
    "depto_id": [1, 2, 1]
})
departamentos = pl.DataFrame({
    "depto_id": [1, 2],
    "nombre_depto": ["Ventas", "IT"]
})

resultado = empleados.join(departamentos, on="depto_id")

Tipos de Join

Controla qué filas se mantienen según la coincidencia entre DataFrames.

# Inner join (solo coincidencias)
df.join(df2, on="clave", how="inner")

# Left join (todo el izquierdo)
df.join(df2, on="clave", how="left")

# Right join (todo el derecho)
df.join(df2, on="clave", how="right")

# Full join (todo)
df.join(df2, on="clave", how="full")

# Cross join (producto cartesiano)
df.join(df2, how="cross")

Join por múltiples columnas

Puedes unir usando varias columnas como clave.

# Join por múltiples columnas
df.join(df2, on=["ciudad", "año"])

# Claves diferentes en cada DataFrame
df.join(df2, left_on="empleado_id", right_on="id")

pl.concat()

Concatena (une) múltiples DataFrames de forma vertical u horizontal.

# Concatenar verticalmente (mismas columnas)
df1 = pl.DataFrame({"producto": ["A", "B"], "ventas": [100, 200]})
df2 = pl.DataFrame({"producto": ["C", "D"], "ventas": [150, 250]})
todos = pl.concat([df1, df2])

# Concatenar horizontalmente (mismas filas)
df_extra = pl.DataFrame({"region": ["Norte", "Sur"]})
resultado = pl.concat([df1, df_extra], how="horizontal")

df.pivot()

Reorganiza datos de formato largo a ancho (similar a tablas dinámicas).

# Pivot básico
ventas = pl.DataFrame({
    "vendedor": ["Ana", "Ana", "Bob", "Bob"],
    "producto": ["A", "B", "A", "B"],
    "monto": [100, 200, 150, 250]
})

pivot = ventas.pivot(
    values="monto",
    index="vendedor",
    columns="producto"
)

df.pivot() con agregación

Pivot con agregaciones personalizadas (sum, mean, etc.).

# Pivot con agregación
pivot_agg = ventas.pivot(
    values="monto",
    index="vendedor",
    columns="producto",
    aggregate_function="sum"
)

df.unpivot()

Convierte un DataFrame de formato ancho a largo (des-apila columnas). Equivalente a melt() en pandas.

# Unpivot (wide to long)
df = pl.DataFrame({
    "producto": ["A", "B"],
    "Q1": [100, 200],
    "Q2": [150, 250]
})

largo = df.unpivot(
    on=["Q1", "Q2"],        # Columnas a des-apilar
    index=["producto"],      # Columnas que se mantienen
    variable_name="trimestre",
    value_name="ventas"
)

Parsear fechas con pl.col().str.to_datetime()

Convierte strings de fechas a tipo datetime.

# Convertir strings a fechas
df = df.with_columns(
    pl.col("fecha_str").str.to_datetime("%Y-%m-%d").alias("fecha")
)

# Formatos comunes
pl.col("fecha").str.to_datetime("%d/%m/%Y")   # 31/12/2024
pl.col("fecha").str.to_datetime("%Y-%m-%d")   # 2024-12-31

Rangos de fechas con pl.date_range()

Genera secuencias de fechas.

from datetime import date

# Crear rango de fechas
rango = pl.date_range(
    start=date(2024, 1, 1),
    end=date(2024, 1, 10),
    interval="1d",
    eager=True
)

# Intervalos: "1d", "1mo", "1y", "1h", "1w"

Componentes de fecha con .dt

Extrae componentes de fechas (año, mes, día, etc.).

# Extraer componentes
df = df.with_columns([
    pl.col("fecha").dt.year().alias("año"),
    pl.col("fecha").dt.month().alias("mes"),
    pl.col("fecha").dt.day().alias("dia"),
    pl.col("fecha").dt.weekday().alias("dia_semana"),
])

Lazy API: df.lazy() y collect()

La API lazy retrasa la ejecución hasta collect(), permitiendo optimizaciones.

# Crear LazyFrame y ejecutar
resultado = (
    df.lazy()
    .filter(pl.col("ventas") > 100)
    .collect()
)

# Ver plan de ejecución
plan = df.lazy().filter(pl.col("ventas") > 100).explain()
print(plan)

Chaining de expresiones

Encadena múltiples operaciones de forma fluida.

resultado = (
    df.lazy()
    .with_columns(
        (pl.col("cantidad") * pl.col("precio")).alias("total")
    )
    .filter(pl.col("total") > 100)
    .sort("total", descending=True)
    .collect()
)
